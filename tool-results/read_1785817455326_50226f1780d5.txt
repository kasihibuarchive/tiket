   256→  function toggleSeat(seatCode: string) {

   264→  function addGaTickets() {
   265→    if (!selectedGaZone || !selectedGaZoneInfo || gaQuantity < 1) return
   266→    const available = selectedGaZoneInfo.seats
   267→    const qty = Math.min(gaQuantity, available.length)
   268→
   269→    // Find existing seat codes for this zone that are already selected
   270→    const existingInZone = selectedSeats.filter((s) => s.startsWith(`${selectedGaZone}-`))
   271→    const existingNumbers = existingInZone.map((s) => {
   272→      const parts = s.split('-')
   273→      return parseInt(parts[parts.length - 1], 10) || 0
   274→    })
   275→    const maxExisting = existingNumbers.length > 0 ? Math.max(...existingNumbers) : 0
   276→
   277→    // Also check existing seat codes in the available seats for this zone to avoid duplicates
   278→    const allExistingCodes = new Set(seats.map((s) => s.seatCode))
   279→    const nextStart = Math.max(maxExisting, ...Array.from(allExistingCodes)
   280→      .filter((c) => c.startsWith(`${selectedGaZone}-`))
   281→      .map((c) => parseInt(c.split('-').pop() || '0', 10) || 0)) + 1
   282→
   283→    const newCodes: string[] = []
   284→    for (let i = 0; i < qty; i++) {
   285→      newCodes.push(`${selectedGaZone}-${nextStart + i}`)
   286→    }
   287→
   288→    setSelectedSeats((prev) => [...prev, ...newCodes])
   289→    setGaQuantity(1)
   290→  }
   291→

   296→  function getSeatColorClass(seat: SeatData): string {
   297→    if (seat.status !== 'AVAILABLE') return 'bg-gray-300 text-gray-500 cursor-not-allowed'

   300→  }

   311→  async function handleSubmit() {
   312→    if (!selectedEventId || !guestName) {
   313→      setSubmitResult({ success: false, message: 'Harap masukkan nama tamu.' })
   314→      return
   315→    }
   316→    if (selectedSeats.length === 0) {
   317→      setSubmitResult({ success: false, message: 'Harap pilih minimal 1 kursi.' })
   318→      return
   319→    }
   320→
   321→    setIsSubmitting(true)
   322→    setSubmitResult(null)
   323→
   324→    try {
   325→      const res = await fetch('/api/admin/tickets/complimentary', {
   326→        method: 'POST',
   327→        headers: {
   328→          'Content-Type': 'application/json',
   329→          ...(adminInfo ? { 'x-admin-id': adminInfo.id, 'x-admin-name': adminInfo.name || adminInfo.role } : {}),
   330→        },
   331→        body: JSON.stringify({
   332→          eventId: selectedEventId,
   333→          seatCodes: selectedSeats,
   334→          guestName,
   335→          guestEmail: '',
   336→          guestPhone: '',
   337→          showDateId: selectedShowDateId || undefined,
   338→        }),
   339→      })
   340→
   341→      const data = await res.json()
   342→
   343→      if (res.ok) {
   344→        setSubmitResult({ success: true, message: `Tiket OTS berhasil dibuat! TRX: ${data.transactionId}` })
   345→        // Reset everything
   346→        setGuestName('')
   347→        setSelectedSeats([])
   348→        setCurrentStep(0)
   349→        setSelectedEventId('')
   350→        // Refresh history
   351→        const recentRes = await fetch('/api/admin/tickets/complimentary')
   352→        if (recentRes.ok) {
   353→          const recentData = await recentRes.json()
   354→          setRecentTickets(recentData.tickets || [])
   355→        }
   356→      } else {
   357→        setSubmitResult({ success: false, message: data.error || 'Gagal mengirim tiket.' })
   358→      }
   359→    } catch {
   360→      setSubmitResult({ success: false, message: 'Terjadi kesalahan jaringan.' })
   361→    } finally {
   362→      setIsSubmitting(false)
   363→    }
   364→  }
