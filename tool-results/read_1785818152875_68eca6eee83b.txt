    81→    81→  // Format promo description for display
    82→    82→  function formatPromoDescription(promo: PromoDetails): string {
    83→    83→    if (promo.discountType === 'BUNDLING_TICKET') {
    84→    84→      return `Bundling: beli ${promo.bundleSize} tiket, diskon Rp ${promo.bundleDiscount.toLocaleString('id-ID')}/bundle`
    85→    85→    }
    86→    86→    const valueStr = promo.discountType === 'PERCENT'
    87→    87→      ? `${promo.discountValue}%`
    88→    88→      : `Rp ${promo.discountValue.toLocaleString('id-ID')}`
    89→    89→    const targetMap: Record<string, string> = {
    90→    90→      ALL: 'semua',
    91→    91→      TICKET: 'tiket',
    92→    92→      MERCH: 'merchandise',
    93→    93→      BUNDLING: 'tiket + merchandise',
    94→    94→    }
    95→    95→    const targetStr = targetMap[promo.target] || promo.target
    96→    96→    return `Diskon ${valueStr} (${targetStr})`
    97→    97→  }

    99→    99→  function parseMerchandise(data: string | null): Array<{ name: string; quantity: number }> {
   100→   100→    if (!data) return []
   101→   101→    try { return JSON.parse(data) } catch { return [] }
   102→   102→  }
   103→   103→
   104→   104→  function parseSeatCodes(codes: string | null): string[] {
   105→   105→    if (!codes) return []
   106→   106→    try { return JSON.parse(codes) } catch {
   107→   107→      return codes.split(',').map((s) => s.trim()).filter(Boolean)
   108→   108→    }
   109→   109→  }
   110→   110→
   111→   111→  function extractTransactionId(qrText: string): string | null {
   112→   112→    const parts = qrText.split('KODE TRX:')
   113→   113→    if (parts.length < 2) return null
   114→   114→    return parts[1].trim()
   115→   115→  }
   116→   116→
   117→   117→  async function performCheckIn(transactionId: string, overrideAction?: 'FORCE_VALID' | 'FORCE_INVALID' | 'RESET_COOLDOWN', overrideNote?: string) {
   118→   118→    setScanState('LOADING')
   119→   119→    setLastTransactionId(transactionId)
   120→   120→    if (overrideAction) setOverrideLoading(overrideAction)
   121→   121→
   122→   122→    try {
   123→   123→      const res = await fetch('/api/check-in', {
   124→   124→        method: 'POST',
   125→   125→        headers: { 'Content-Type': 'application/json' },
   126→   126→        body: JSON.stringify({
   127→   127→          transactionId,
   128→   128→          overrideAction,
   129→   129→          overrideNote: overrideNote || undefined,
   130→   130→        }),
   131→   131→      })
   132→   132→
   133→   133→      const data = await res.json()
   134→   134→
   135→   135→      if (data.status === 'SUCCESS') {
   136→   136→        setScanState('SUCCESS')
   137→   137→      } else if (data.status === 'WARNING') {
   138→   138→        setScanState('WARNING')
   139→   139→      } else {
   140→   140→        setScanState('ERROR')
   141→   141→      }
   142→   142→
   143→   143→      setResult(data)
   144→   144→      // Track which day was targeted so the override UI can hint at it
   145→   145→      setLastTargetDayId(data.transaction?.festival?.targetShowDateId || null)
   146→   146→    } catch (err) {
   147→   147→      setScanState('ERROR')
   148→   148→      setResult({
   149→   149→        status: 'ERROR',
   150→   150→        message: 'Gagal terhubung ke server. Silakan coba lagi.',
   151→   151→      })
   152→   152→    } finally {
   153→   153→      setOverrideLoading(null)
   154→   154→    }
   155→   155→  }
   156→   156→
   157→   157→  function handleScanSuccess(decodedText: string) {
   158→   158→    const transactionId = extractTransactionId(decodedText)
   159→   159→    if (!transactionId) {
   160→   160→      setScanState('ERROR')
   161→   161→      setResult({
   162→   162→        status: 'ERROR',
   163→   163→        message: 'Format QR tidak valid. Pastikan QR tiket yang benar.',
   164→   164→      })
   165→   165→      return
   166→   166→    }
   167→   167→    stopScanner()
   168→   168→    performCheckIn(transactionId)
   169→   169→  }

   171→   171→  async function startScanner() {
   172→   172→    await stopScanner()

   174→   174→      const { Html5Qrcode } = await import('html5-qrcode')
   175→   175→      const scanner = new Html5Qrcode('qr-reader')
   176→   176→      await scanner.start(
   177→   177→        { facingMode: 'environment' },
   178→   178→        {
   179→   179→          fps: 10,
   180→   180→          qrbox: { width: 280, height: 280 },
   181→   181→          aspectRatio: 1.0,
   182→   182→        },
   183→   183→        (decodedText: string) => handleScanSuccess(decodedText),
   184→   184→        () => { /* QR not found in frame - ignore */ }
   185→   185→      )
   186→   186→      scannerRef.current = scanner
   187→   187→      setScanState('SCANNING')
   188→   188→    } catch (err) {
   189→   189→      console.error('Failed to start scanner:', err)
   190→   190→      setScanState('ERROR')
   191→   191→      setResult({
   192→   192→        status: 'ERROR',
   193→   193→        message: 'Gagal mengakses kamera. Pastikan izin kamera diberikan.',
   194→   194→      })
   195→   195→    }
   196→   196→  }
   197→   197→
   198→   198→  async function stopScanner() {
   199→   199→    if (scannerRef.current) {
   200→   200→      try {
   201→   201→        const state = scannerRef.current.getState()

   204→   204→      } catch { /* ignore */ }
   205→   205→      scannerRef.current = null
   206→   206→    }
   207→   207→  }
