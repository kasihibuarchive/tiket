   345→  async function handleSubmit() {
   346→    if (!selectedEventId || !guestName || !guestEmail || !guestPhone) {
   347→      setSubmitResult({ success: false, message: 'Harap lengkapi semua field yang wajib diisi.' })
   348→      return
   349→    }
   350→
   351→    let seatCodes: string[]
   352→
   353→    if (isNumberedSeatMap) {
   354→      if (selectedSeats.length === 0) {
   355→        setSubmitResult({ success: false, message: 'Harap pilih minimal 1 kursi.' })
   356→        return
   357→      }
   358→      seatCodes = selectedSeats
   359→    } else {
   360→      if (gaQuantity < 1) {
   361→        setSubmitResult({ success: false, message: 'Jumlah tiket minimal 1.' })
   362→        return
   363→      }
   364→      // Generate GA seat codes
   365→      seatCodes = Array.from({ length: gaQuantity }, (_, i) =>
   366→        gaZone ? `${gaZone}-${i + 1}` : `GA-${i + 1}`
   367→      )
   368→    }
   369→
   370→    setIsSubmitting(true)
   371→    setSubmitResult(null)
   372→
   373→    try {
   374→      const res = await fetch('/api/admin/tickets/complimentary', {
   375→        method: 'POST',
   376→        headers: { 'Content-Type': 'application/json' },
   377→        body: JSON.stringify({
   378→          eventId: selectedEventId,
   379→          seatCodes,
   380→          guestName,
   381→          guestEmail,
   382→          guestPhone,
   383→          showDateId: selectedShowDateId || undefined,
   384→        }),
   385→      })
   386→
   387→      const data = await res.json()
   388→
   389→      if (res.ok) {
   390→        setSubmitResult({
   391→          success: true,
   392→          message: `Tiket komplimen berhasil dibuat! Transaction ID: ${data.transactionId}`,
   393→        })
   394→        // Reset form
   395→        setGuestName('')
   396→        setGuestEmail('')
   397→        setGuestPhone('')
   398→        setSelectedSeats([])
   399→        setGaQuantity(1)
   400→        setGaZone('')
   401→        // Refresh recent tickets
   402→        const recentRes = await fetch('/api/admin/tickets/complimentary')
   403→        if (recentRes.ok) {
   404→          const recentData = await recentRes.json()
   405→          setRecentTickets(recentData.tickets || [])
   406→          setTotalComplimentary(recentData.total || 0)
   407→        }
   408→      } else {
   409→        setSubmitResult({
   410→          success: false,
   411→          message: data.error || 'Gagal membuat tiket komplimen.',
   412→        })
   413→      }
   414→    } catch (err) {
   415→      setSubmitResult({
   416→        success: false,
   417→        message: 'Terjadi kesalahan jaringan. Silakan coba lagi.',
   418→      })
   419→    } finally {
   420→      setIsSubmitting(false)
   421→    }
   422→  }
