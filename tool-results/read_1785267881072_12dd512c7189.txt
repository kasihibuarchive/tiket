   133→  async function handleZoneLock(zoneName: string) {
   134→    if (!confirm(`Kunci zona "${zoneName}"?\nSemua kursi tersedia akan ditutup dari penjualan.`)) return
   135→    setIsZoneLocking(prev => ({ ...prev, [zoneName]: true }))
   136→    try {
   137→      const res = await fetch(`/api/admin/events/${eventId}/zone-lock`, {
   138→        method: 'POST',
   139→        headers: { 'Content-Type': 'application/json' },
   140→        body: JSON.stringify({ zoneName }),
   141→      })
   142→      const data = await res.json()
   143→      if (res.ok) {
   144→        alert(data.message)
   145→        onRefresh?.()
   146→      } else {
   147→        alert(data.error || 'Gagal mengunci zona')
   148→      }
   149→    } catch {
   150→      alert('Gagal mengunci zona')
   151→    } finally {
   152→      setIsZoneLocking(prev => ({ ...prev, [zoneName]: false }))
   153→    }
   154→  }

   156→  async function handleZoneUnlock(zoneName: string) {
   157→    if (!confirm(`Buka kunci zona "${zoneName}"?\nSemua kursi yang ditutup akan tersedia kembali.`)) return
   158→    setIsZoneLocking(prev => ({ ...prev, [zoneName]: true }))
   159→    try {
   160→      const res = await fetch(`/api/admin/events/${eventId}/zone-lock?zoneName=${encodeURIComponent(zoneName)}`, {
   161→        method: 'DELETE',
   162→      })
   163→      const data = await res.json()
   164→      if (res.ok) {
   165→        alert(data.message)
   166→        onRefresh?.()
   167→      } else {
   168→        alert(data.error || 'Gagal membuka zona')
   169→      }
   170→    } catch {
   171→      alert('Gagal membuka zona')
   172→    } finally {
   173→      setIsZoneLocking(prev => ({ ...prev, [zoneName]: false }))
   174→    }
   175→  }
   176→  async function handleReserve(zoneName: string) {
   177→    const qty = reserveQty[zoneName] || 0
   178→    if (qty < 1) {
   179→      alert('Masukkan jumlah slot yang ingin direservasi')
   180→      return
   181→    }
   182→    setIsReserving(prev => ({ ...prev, [zoneName]: true }))
   183→    try {
   184→      const res = await fetch(`/api/admin/events/${eventId}/reserve-invitation`, {
   185→        method: 'POST',
   186→        headers: { 'Content-Type': 'application/json' },
   187→        body: JSON.stringify({ zoneName, quantity: qty }),
   188→      })
   189→      const data = await res.json()
   190→      if (res.ok) {
   191→        alert(data.message)
   192→        setReserveQty(prev => ({ ...prev, [zoneName]: 0 }))
   193→        onRefresh?.()
   194→      } else {
   195→        alert(data.error || 'Gagal reservasi slot undangan')
   196→      }
   197→    } catch {
   198→      alert('Gagal reservasi slot undangan')
   199→    } finally {
   200→      setIsReserving(prev => ({ ...prev, [zoneName]: false }))
   201→    }
   202→  }

   230→  // ─── Image upload handler ───────────────────────────────────────────
   231→  async function handleImageUpload(e: React.ChangeEvent<HTMLInputElement>) {
   232→    const file = e.target.files?.[0]
   233→    if (!file) return
   234→
   235→    // Check file size (max 5MB)
   236→    if (file.size > 5 * 1024 * 1024) {
   237→      alert('Ukuran file maksimal 5MB')
   238→      return
   239→    }
   240→
   241→    const reader = new FileReader()
   242→    reader.onload = async (ev) => {
   243→      const base64 = ev.target?.result as string
   244→      setLayoutImagePreview(base64)
   245→
   246→      setIsUploading(true)
   247→      try {
   248→        const res = await fetch(`/api/admin/events/${eventId}`, {
   249→          method: 'PUT',
   250→          headers: { 'Content-Type': 'application/json' },
   251→          body: JSON.stringify({ layoutImage: base64 }),
   252→        })
   253→        if (res.ok) {
   254→          setLayoutImage(base64)
   255→        } else {
   256→          alert('Gagal mengupload gambar')
   257→          setLayoutImagePreview(layoutImage)
   258→        }
   259→      } catch {
   260→        alert('Gagal mengupload gambar')
   261→        setLayoutImagePreview(layoutImage)
   262→      } finally {
   263→        setIsUploading(false)
   264→      }
   265→    }
   266→    reader.readAsDataURL(file)
   267→  }

   312→  // ─── Sync zones from price categories ───────────────────────────────
   313→  // Rebuilds gaZonesDef from current priceCategories. Preserves notes &
   314→  // capacity for zones that already exist (matched by name).
   315→  function handleSyncFromCategories() {
   316→    if (priceCategories.length === 0) {
   317→      alert('Belum ada kategori harga di event ini. Tambahkan kategori di halaman Events terlebih dahulu.')
   318→      return
   319→    }
   320→    const confirmMsg = `Sync zona dari ${priceCategories.length} kategori harga?\n\n` +
   321→      `• Zona yang sudah ada: harga & warna diperbarui, KAPASITAS DIPERTAHANKAN sesuai adjust lo\n` +
   322→      `• Zona baru: dibuat dengan kapasitas default 100 (bisa di-adjust setelah sync)\n` +
   323→      `• Zona yang gak match kategori mana pun: tetap dipertahankan\n\n` +
   324→      `Kategori:\n${priceCategories.map(pc => `• ${pc.name} — Rp ${pc.price.toLocaleString('id-ID')}`).join('\n')}`
   325→    if (!confirm(confirmMsg)) return
   326→
   327→    const next: GaZoneDef[] = []
   328→    const usedZoneNames = new Set<string>()
   329→
   330→    // 1. For each price category, ensure a matching zone exists
   331→    for (const pc of priceCategories) {
   332→      const existing = gaZonesDef.find(z =>
   333→        z.name.toLowerCase() === pc.name.toLowerCase() ||
   334→        (z.priceCategoryName || '').toLowerCase() === pc.name.toLowerCase()
   335→      )
   336→      next.push({
   337→        name: pc.name,
   338→        capacity: existing?.capacity ?? 100,
   339→        price: pc.price,
   340→        color: pc.colorCode,
   341→        priceCategoryName: pc.name,
   342→        notes: existing?.notes || '',
   343→      })
   344→      usedZoneNames.add(pc.name.toLowerCase())
   345→    }
   346→
   347→    // 2. Keep zones that don't match any price category (admin-added custom zones)
   348→    for (const z of gaZonesDef) {
   349→      const matched = priceCategories.some(pc =>
   350→        pc.name.toLowerCase() === z.name.toLowerCase() ||
   351→        pc.name.toLowerCase() === (z.priceCategoryName || '').toLowerCase()
   352→      )
   353→      if (!matched) next.push(z)
   354→    }
   355→
   356→    setGaZonesDef(next)
   357→    alert(`✓ Synced ${priceCategories.length} zona dari kategori harga. Klik "Simpan Zona" untuk menyimpan.`)
   358→  }
