   388→  async function handleSubmit() {
   389→    if (!selectedEventId || !guestName) {
   390→      setSubmitResult({ success: false, message: 'Harap isi nama event dan nama tamu.' })
   391→      return
   392→    }
   393→    // Email/Phone optional untuk OTS walk-in (backend support empty email)
   394→    // Tapi kalau email diisi, validasi format
   395→    if (guestEmail && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(guestEmail)) {
   396→      setSubmitResult({ success: false, message: 'Format email tidak valid.' })
   397→      return
   398→    }
   399→
   400→    let seatCodes: string[]
   401→    let festivalPackage: { priceCategoryId: string; quantity: number } | undefined
   402→
   403→    if (isFestival) {
   404→      // ─── FESTIVAL MODE: pick package + quantity ────────────────────
   405→      if (!selectedFestivalPkg) {
   406→        setSubmitResult({ success: false, message: 'Pilih paket festival terlebih dahulu.' })
   407→        return
   408→      }
   409→      if (festivalQuantity < 1) {
   410→        setSubmitResult({ success: false, message: 'Jumlah tiket minimal 1.' })
   411→        return
   412→      }
   413→      const avail = festivalAvailability[selectedFestivalPkg.id] ?? 0
   414→      if (festivalQuantity > avail) {
   415→        setSubmitResult({
   416→          success: false,
   417→          message: `Tiket tersedia hanya ${avail}. Kurangi jumlah atau pilih paket lain.`,
   418→        })
   419→        return
   420→      }
   421→      // Seat codes: use package name as prefix (matches zoneName in seat table)
   422→      seatCodes = Array.from({ length: festivalQuantity }, (_, i) =>
   423→        `${selectedFestivalPkg.name}-COMP${i + 1}`
   424→      )
   425→      festivalPackage = {
   426→        priceCategoryId: selectedFestivalPkg.id,
   427→        quantity: festivalQuantity,
   428→      }
   429→    } else if (isNumberedSeatMap) {
   430→      if (selectedSeats.length === 0) {
   431→        setSubmitResult({ success: false, message: 'Harap pilih minimal 1 kursi.' })
   432→        return
   433→      }
   434→      seatCodes = selectedSeats
   435→    } else {
   436→      if (gaQuantity < 1) {
   437→        setSubmitResult({ success: false, message: 'Jumlah tiket minimal 1.' })
   438→        return
   439→      }
   440→      // Generate GA seat codes
   441→      seatCodes = Array.from({ length: gaQuantity }, (_, i) =>
   442→        gaZone ? `${gaZone}-${i + 1}` : `GA-${i + 1}`
   443→      )
   444→    }
   445→
   446→    setIsSubmitting(true)
   447→    setSubmitResult(null)
   448→
   449→    try {
   450→      const res = await fetch('/api/admin/tickets/complimentary', {
   451→        method: 'POST',
   452→        headers: { 'Content-Type': 'application/json' },
   453→        body: JSON.stringify({
   454→          eventId: selectedEventId,
   455→          seatCodes,
   456→          guestName,
   457→          guestEmail,
   458→          guestPhone,
   459→          showDateId: selectedShowDateId || undefined,
   460→          festivalPackage,
   461→        }),
   462→      })
   463→
   464→      const data = await res.json()
   465→
   466→      if (res.ok) {
   467→        setSubmitResult({
   468→          success: true,
   469→          message: `Tiket komplimen berhasil dibuat! Transaction ID: ${data.transactionId}`,
   470→        })
   471→        // Reset form
   472→        setGuestName('')
   473→        setGuestEmail('')
   474→        setGuestPhone('')
   475→        setSelectedSeats([])
   476→        setGaQuantity(1)
   477→        setGaZone('')
   478→        setFestivalQuantity(1)
   479→        setSelectedFestivalPkgId('')
   480→        // Refresh recent tickets
   481→        const recentRes = await fetch('/api/admin/tickets/complimentary')
   482→        if (recentRes.ok) {
   483→          const recentData = await recentRes.json()
   484→          setRecentTickets(recentData.tickets || [])
   485→          setTotalComplimentary(recentData.total || 0)
   486→        }
   487→      } else {
   488→        setSubmitResult({
   489→          success: false,
   490→          message: data.error || 'Gagal membuat tiket komplimen.',
   491→        })
   492→      }
   493→    } catch (err) {
   494→      setSubmitResult({
   495→        success: false,
   496→        message: 'Terjadi kesalahan jaringan. Silakan coba lagi.',
   497→      })
   498→    } finally {
   499→      setIsSubmitting(false)
   500→    }
   501→  }
