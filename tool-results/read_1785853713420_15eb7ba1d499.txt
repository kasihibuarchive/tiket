    56→    56→    56→    })

    91→    91→    91→        try { applicableDayIds = JSON.parse(priceCat.applicableDayIds) } catch { /* ignore */ }
    92→    92→    92→      }
    93→    93→    93→
    94→    94→    94→      if (applicableDayIds.length === 0) {

    97→    97→    97→

    99→    99→    99→      // Multi-day access is encoded in the seatCode stored on the transaction:
   100→   100→   100→      //   "VIP-1@day1abc,day2def,day3ghi"
   101→   101→   101→      // The seat itself has eventShowDateId = null. The applicable days are
   102→   102→   102→      // derived from the price category's applicableDayIds at checkout time.
   103→   103→   103→      //
   104→   104→   104→      // Availability = count of AVAILABLE seats in this package's zone, period.
   105→   105→   105→      // No per-day breakdown, no min-across-days.

   107→   107→   107→
   108→   108→   108→      // Find `qty` available GA seats in this price category's zone
   109→   109→   109→      // (zoneName === price category name in our auto-built GA config)

   111→   111→   111→        where: {
   112→   112→   112→          eventId,
   113→   113→   113→          priceCategoryId: priceCat.id,
   114→   114→   114→          status: 'AVAILABLE',
   115→   115→   115→        },
   116→   116→   116→        orderBy: { seatCode: 'asc' },
   117→   117→   117→        take: qty,
   118→   118→   118→      })
   119→   119→   119→
   120→   120→   120→      if (availableSeats.length < qty) {
   121→   121→   121→        return NextResponse.json({
   122→   122→   122→          error: `Tiket tinggal ${availableSeats.length} (Anda pesan ${qty}). Kurangi jumlah atau pilih paket lain.`,
   123→   123→   123→        }, { status: 409 })
   124→   124→   124→      }
   125→   125→   125→
   126→   126→   126→      for (const s of availableSeats) {
   127→   127→   127→        allPickedSeats.push({
   128→   128→   128→          id: s.id,
   129→   129→   129→          seatCode: s.seatCode,
   130→   130→   130→          priceCategoryId: s.priceCategoryId!,
   131→   131→   131→        })
   132→   132→   132→      }
   133→   133→   133→
   134→   134→   134→      // Lock the picked seats under this checkout session
   135→   135→   135→      const lockedUntil = new Date(Date.now() + 10 * 60 * 1000)

   137→   137→   137→        where: { id: { in: allPickedSeats.map(s => s.id) } },

   139→   139→   139→      })
   140→   140→   140→
   141→   141→   141→      festivalAutoSeats = allPickedSeats as any
   142→   142→   142→      // Encode applicable day IDs into seatCode string for check-in:
   143→   143→   143→      // "VIP-1@day1abc,day2def,day3ghi" — usher scans, system knows ticket
   144→   144→   144→      // is valid for any of those days.
   145→   145→   145→      const dayIdsJoined = applicableDayIds.join(',')
   146→   146→   146→      festivalSeatCodes = allPickedSeats.map(s => `${s.seatCode}@${dayIdsJoined}`)
   147→   147→   147→    } else {
   148→   148→   148→      // REGULAR Mode: validate seats as before
   149→   149→   149→      const seatWhere: any = { eventId, seatCode: { in: seatCodes } }
   150→   150→   150→      if (showDateId) {
   151→   151→   151→        seatWhere.eventShowDateId = showDateId
   152→   152→   152→      }
   153→   153→   153→      const seats = await db.seat.findMany({ where: seatWhere })
   154→   154→   154→      if (seats.length !== seatCodes.length) return NextResponse.json({ error: 'Kursi tidak ditemukan' }, { status: 404 })
   155→   155→   155→
   156→   156→   156→      const invalidSeats = seats.filter((s) => s.status === 'SOLD')

   158→   158→   158→        return NextResponse.json({ error: 'Kursi ' + invalidSeats.map((s) => s.seatCode).join(', ') + ' sudah terjual' }, { status: 409 })
   159→   159→   159→      }
   160→   160→   160→
   161→   161→   161→      const notOurs = seats.filter((s) => s.status === 'LOCKED_TEMPORARY' && s.lockedBy !== checkoutId)
   162→   162→   162→      if (notOurs.length > 0) {
   163→   163→   163→        return NextResponse.json({ error: 'Kursi ' + notOurs.map((s) => s.seatCode).join(', ') + ' sedang dipilih orang lain.' }, { status: 409 })
   164→   164→   164→      }
   165→   165→   165→
   166→   166→   166→      // Re-lock seats
   167→   167→   167→      const lockedUntil = new Date(Date.now() + 10 * 60 * 1000)
   168→   168→   168→      await db.seat.updateMany({
   169→   169→   169→        where: { eventId, seatCode: { in: seatCodes }, lockedBy: checkoutId },
   170→   170→   170→        data: { status: 'LOCKED_TEMPORARY', lockedUntil, lockedBy: checkoutId },
   171→   171→   171→      })
   172→   172→   172→    }
   173→   173→   173→
   174→   174→   174→    // Resolve payment method — accept Tripay channel codes or legacy QRIS/NON_QRIS
   175→   175→   175→    let resolvedMethod = paymentMethod || 'BCAVA'
   176→   176→   176→    if (LEGACY_METHOD_MAP[resolvedMethod]) {
   177→   177→   177→      resolvedMethod = LEGACY_METHOD_MAP[resolvedMethod]
   178→   178→   178→    }
   179→   179→   179→
   180→   180→   180→    // Calculate seat prices & items

   183→   183→   183→    let seatTotal = 0
   184→   184→   184→    const items: any[] = []
   185→   185→   185→
   186→   186→   186→    if (isFestivalCheckout) {
   187→   187→   187→      // Festival Mode: each ticket = 1 festival package (covers all applicable days)
   188→   188→   188→      // We charge `quantity` × package price (not per-day seats)
   189→   189→   189→      const pkg = festivalPackage
   190→   190→   190→      const qty = festivalTicketCount
   191→   191→   191→      const pkgPrice = pkg.unitPrice
   192→   192→   192→      seatTotal = pkgPrice * qty
   193→   193→   193→
   194→   194→   194→      items.push({
   195→   195→   195→        id: `FESTIVAL-${pkg.priceCategoryId}`,
   196→   196→   196→        price: pkgPrice,
   197→   197→   197→        quantity: qty,

   199→   199→   199→        category: 'Tiket Festival',
   200→   200→   200→        priceCategoryId: pkg.priceCategoryId,
   201→   201→   201→      })
   202→   202→   202→    } else {
   203→   203→   203→      // REGULAR Mode: use seat prices as before
   204→   204→   204→      const seatWhere: any = { eventId, seatCode: { in: seatCodes } }
   205→   205→   205→      if (showDateId) {
   206→   206→   206→        seatWhere.eventShowDateId = showDateId
   207→   207→   207→      }
   208→   208→   208→      const seatPrices = await db.seat.findMany({ where: seatWhere, select: { seatCode: true, priceCategoryId: true } })
   209→   209→   209→
   210→   210→   210→      for (const s of seatPrices) {
   211→   211→   211→        const cat = priceCats.find((p) => p.id === s.priceCategoryId)
   212→   212→   212→        if (!cat) return NextResponse.json({ error: 'Harga kursi belum diatur' }, { status: 400 })
   213→   213→   213→        seatTotal += cat.price
   214→   214→   214→        items.push({ id: s.seatCode, price: cat.price, quantity: 1, name: 'Kursi ' + s.seatCode, category: 'Tiket', priceCategoryId: s.priceCategoryId })
   215→   215→   215→      }
   216→   216→   216→    }
   217→   217→   217→
   218→   218→   218→    // Admin fee — flat per ticket (festival uses festivalTicketCount)
   219→   219→   219→    const adminFeePerTicket = event?.adminFee || 0
   220→   220→   220→    const effectiveTicketCount = isFestivalCheckout ? festivalTicketCount : seatCodes.length
   221→   221→   221→    const adminFeeTotal = adminFeePerTicket * effectiveTicketCount
   222→   222→   222→
   223→   223→   223→    if (adminFeeTotal > 0) {
   224→   224→   224→      items.push({ id: 'ADMIN-FEE', price: adminFeeTotal, quantity: 1, name: 'Biaya Admin', category: 'Biaya' })
   225→   225→   225→    }
   226→   226→   226→
   227→   227→   227→    // Promo code - calculate discount (using Jakarta timezone)

   232→   232→   232→    }
   233→   233→   233→

   235→   235→   235→    let promoCodeData: Awaited<ReturnType<typeof db.promoCode.findUnique>> | null = null
   236→   236→   236→    let promoTarget = 'ALL'

   240→   240→   240→      if (promoCodeData && promoCodeData.isActive && promoCodeData.currentUses < promoCodeData.maxUses) {

   246→   246→   246→        if (nowJakarta >= fromJakarta && nowJakarta <= untilJakarta) {

   249→   249→   249→          // Use effectiveTicketCount so festival mode (where seatCodes is empty) still validates correctly

   251→   251→   251→
   252→   252→   252→          if (ticketCountForPromo < (promoCodeData.minTickets || 0)) {
   253→   253→   253→            return NextResponse.json({ error: `Promo ini berlaku untuk pembelian minimal ${promoCodeData.minTickets} tiket` }, { status: 400 })
   254→   254→   254→          }
   255→   255→   255→          if (!hasMerch && (promoCodeData.minMerchItems || 0) > 0) {
   256→   256→   256→            return NextResponse.json({ error: `Promo ini berlaku jika membeli minimal ${promoCodeData.minMerchItems} merchandise` }, { status: 400 })
   257→   257→   257→          }
   258→   258→   258→
   259→   259→   259→          if (promoTarget === 'BUNDLING' && !(ticketCountForPromo > 0 && hasMerch)) {
   260→   260→   260→            return NextResponse.json({ error: 'Promo bundling hanya berlaku jika membeli tiket + merchandise' }, { status: 400 })
   261→   261→   261→          }

   263→   263→   263→            return NextResponse.json({ error: 'Promo ini hanya berlaku untuk merchandise' }, { status: 400 })
   264→   264→   264→          }
   265→   265→   265→
   266→   266→   266→          // --- Zone restriction validation ---

   270→   270→   270→          if (promoCodeData.applicableZoneNames) {

   272→   272→   272→              const allowedZones: string[] = JSON.parse(promoCodeData.applicableZoneNames)

   285→   285→   285→                    where: seatWhere,

   287→   287→   287→                  })

   296→   296→   296→                    .filter(Boolean) as string[]
   297→   297→   297→                  allBuyerZones = [...new Set([...buyerZoneNames, ...buyerCategoryNames])]

   299→   299→   299→
   300→   300→   300→                const matchingZones = allBuyerZones.filter((name) => allowedZones.includes(name))

   302→   302→   302→                  return NextResponse.json({ error: `Promo ini hanya berlaku untuk zona: ${allowedZones.join(', ')}` }, { status: 400 })
   303→   303→   303→                }
   304→   304→   304→              }
   305→   305→   305→            } catch {
   306→   306→   306→              // If JSON parse fails, ignore category restriction
   307→   307→   307→            }
   308→   308→   308→          }
   309→   309→   309→        } else {
   310→   310→   310→          // Promo date invalid — clear promo so it doesn't get applied
   311→   311→   311→          promoCodeData = null
   312→   312→   312→        }
   313→   313→   313→      } else {
   314→   314→   314→        // Promo inactive or max uses reached — clear promo
   315→   315→   315→        promoCodeData = null
   316→   316→   316→      }
   317→   317→   317→    }
   318→   318→   318→
   319→   319→   319→    // Merchandise - use DB price, not client-sent price
   320→   320→   320→    let merchDataToSave: any = null
   321→   321→   321→    if (merchandise && Array.isArray(merchandise) && merchandise.length > 0) {

   332→   332→   332→        }
   333→   333→   333→
   334→   334→   334→        const subtotal = merchItem.price * merch.quantity
   335→   335→   335→        merchTotal += subtotal
   336→   336→   336→
   337→   337→   337→        merchDataToSave.push({
   338→   338→   338→          merchandiseId: merch.merchandiseId,
   339→   339→   339→          name: merchItem.name,
   340→   340→   340→          price: merchItem.price,
   341→   341→   341→          quantity: merch.quantity,
   342→   342→   342→        })
   343→   343→   343→
   344→   344→   344→        items.push({
   345→   345→   345→          id: 'MERCH-' + merch.merchandiseId,
   346→   346→   346→          price: merchItem.price,
   347→   347→   347→          quantity: merch.quantity,
   348→   348→   348→          name: merchItem.name,
   349→   349→   349→          category: 'Merchandise',
   350→   350→   350→        })
   351→   351→   351→      }
   352→   352→   352→
   353→   353→   353→      for (const merch of merchandise) {
   354→   354→   354→        await db.merchandise.update({
   355→   355→   355→          where: { id: merch.merchandiseId },
   356→   356→   356→          data: { stock: { decrement: merch.quantity } },
   357→   357→   357→        })
   358→   358→   358→      }

   361→   361→   361→    const merchTotalCalc = merchDataToSave ? merchDataToSave.reduce((s: number, m: any) => s + m.price * m.quantity, 0) : 0

   363→   363→   363→    // ── Calculate discount AFTER merchandise is resolved ──
   364→   364→   364→    if (promoCodeData) {

   373→   373→   373→        if (bundleCount > 0) {

   375→   375→   375→        }
   376→   376→   376→      } else if (promoTarget === 'TICKET') {
   377→   377→   377→        if (isPerItem && ticketCount > 0) {
   378→   378→   378→          const perItemDiscount =
   379→   379→   379→            promoCodeData.discountType === 'PERCENT'
   380→   380→   380→              ? Math.round((ticketSubtotal / ticketCount) * promoCodeData.discountValue / 100)
   381→   381→   381→              : Math.min(promoCodeData.discountValue, ticketSubtotal / ticketCount)
   382→   382→   382→          discountAmount = perItemDiscount * ticketCount
   383→   383→   383→        } else {
   384→   384→   384→          discountAmount =
   385→   385→   385→            promoCodeData.discountType === 'PERCENT'
   386→   386→   386→              ? Math.round(ticketSubtotal * promoCodeData.discountValue / 100)
   387→   387→   387→              : Math.min(promoCodeData.discountValue, ticketSubtotal)
   388→   388→   388→        }
   389→   389→   389→      } else if (promoTarget === 'MERCH' && merchTotalCalc > 0) {
   390→   390→   390→        const totalMerchQty = merchDataToSave ? merchDataToSave.reduce((s: number, m: any) => s + m.quantity, 0) : 0
   391→   391→   391→        if (isPerItem && totalMerchQty > 0) {
   392→   392→   392→          const perItemDiscount =
   393→   393→   393→            promoCodeData.discountType === 'PERCENT'
   394→   394→   394→              ? Math.round((merchTotalCalc / totalMerchQty) * promoCodeData.discountValue / 100)
   395→   395→   395→              : Math.min(promoCodeData.discountValue, merchTotalCalc / totalMerchQty)
   396→   396→   396→          discountAmount = perItemDiscount * totalMerchQty
   397→   397→   397→        } else {
   398→   398→   398→          discountAmount =
   399→   399→   399→            promoCodeData.discountType === 'PERCENT'
   400→   400→   400→              ? Math.round(merchTotalCalc * promoCodeData.discountValue / 100)
   401→   401→   401→              : Math.min(promoCodeData.discountValue, merchTotalCalc)
   402→   402→   402→        }
   403→   403→   403→      } else if (promoTarget === 'ALL' || promoTarget === 'BUNDLING') {
   404→   404→   404→        const targetSubtotal = ticketSubtotal + merchTotalCalc
   405→   405→   405→        const totalItems = ticketCount + (merchDataToSave ? merchDataToSave.reduce((s: number, m: any) => s + m.quantity, 0) : 0)
   406→   406→   406→
   407→   407→   407→        if (isPerItem && totalItems > 0) {
   408→   408→   408→          const perItemDiscount =
   409→   409→   409→            promoCodeData.discountType === 'PERCENT'
   410→   410→   410→              ? Math.round((targetSubtotal / totalItems) * promoCodeData.discountValue / 100)
   411→   411→   411→              : Math.min(promoCodeData.discountValue, targetSubtotal / totalItems)
   412→   412→   412→          discountAmount = perItemDiscount * totalItems
   413→   413→   413→        } else {
   414→   414→   414→          discountAmount =
   415→   415→   415→            promoCodeData.discountType === 'PERCENT'
   416→   416→   416→              ? Math.round(targetSubtotal * promoCodeData.discountValue / 100)
   417→   417→   417→              : Math.min(promoCodeData.discountValue, targetSubtotal)
   418→   418→   418→        }
   419→   419→   419→      }
   420→   420→   420→    }
   421→   421→   421→
   422→   422→   422→    // Final total = seats + admin fee + merch - discount
   423→   423→   423→    const totalAmount = Math.max(seatTotal + adminFeeTotal + merchTotalCalc - discountAmount, 1)
   424→   424→   424→
   425→   425→   425→    console.log('[checkout] seatTotal:', seatTotal, 'adminFee:', adminFeeTotal, 'discount:', discountAmount, 'merchTotal:', merchTotalCalc, 'totalAmount:', totalAmount)

   448→   448→   448→        { error: 'Payment gateway belum dikonfigurasi. Hubungi admin untuk mengatur API key Tripay.' },
   449→   449→   449→        { status: 503 }
   450→   450→   450→      )
   451→   451→   451→    }
   452→   452→   452→
   453→   453→   453→    const expiredTime = Math.floor(Date.now() / 1000) + (24 * 60 * 60) // 24 hours
   454→   454→   454→    const signature = createTransactionSignature(tid, totalAmount)
   455→   455→   455→    const appUrl = process.env.NEXT_PUBLIC_APP_URL || 'http://localhost:3000'
   456→   456→   456→
   457→   457→   457→    // Build order_items for Tripay — always itemized so admin fee is visible
   458→   458→   458→    let orderItems: { sku: string; name: string; price: number; quantity: number }[]
   459→   459→   459→
   460→   460→   460→    if (discountAmount === 0) {

   467→   467→   467→    } else {

   469→   469→   469→      const totalItemValue = items.reduce((sum, item) => sum + item.price * item.quantity, 0)
   470→   470→   470→      let remainingDiscount = discountAmount

   478→   478→   478→          itemDiscount = remainingDiscount
   479→   479→   479→        } else {
   480→   480→   480→          itemDiscount = Math.round(discountAmount * (itemSubtotal / totalItemValue))
   481→   481→   481→          remainingDiscount -= itemDiscount
   482→   482→   482→        }
   483→   483→   483→
   484→   484→   484→        const discountedPrice = Math.max(itemSubtotal - itemDiscount, 1)
   485→   485→   485→        return {

   487→   487→   487→          name: item.name + (itemDiscount > 0 ? ' (diskon)' : ''),

   489→   489→   489→          quantity: item.quantity,
   490→   490→   490→        }
   491→   491→   491→      })
