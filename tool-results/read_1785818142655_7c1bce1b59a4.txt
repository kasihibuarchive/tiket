    70→export default function UsherScannerPage() {
    71→  const [scanState, setScanState] = useState<ScanResultState>('SCANNING')

    73→  const [lastTransactionId, setLastTransactionId] = useState<string | null>(null)
    74→  const [lastTargetDayId, setLastTargetDayId] = useState<string | null>(null)
    75→  const [overrideLoading, setOverrideLoading] = useState<'FORCE_VALID' | 'FORCE_INVALID' | 'RESET_COOLDOWN' | null>(null)

    77→  const [overrideNote, setOverrideNote] = useState('')
    78→  const scannerRef = useRef<any>(null)
    79→  const scannerContainerRef = useRef<HTMLDivElement>(null)

    81→  // Format promo description for display
    82→  function formatPromoDescription(promo: PromoDetails): string {
    83→    if (promo.discountType === 'BUNDLING_TICKET') {
    84→      return `Bundling: beli ${promo.bundleSize} tiket, diskon Rp ${promo.bundleDiscount.toLocaleString('id-ID')}/bundle`
    85→    }
    86→    const valueStr = promo.discountType === 'PERCENT'
    87→      ? `${promo.discountValue}%`
    88→      : `Rp ${promo.discountValue.toLocaleString('id-ID')}`
    89→    const targetMap: Record<string, string> = {
    90→      ALL: 'semua',
    91→      TICKET: 'tiket',
    92→      MERCH: 'merchandise',
    93→      BUNDLING: 'tiket + merchandise',
    94→    }
    95→    const targetStr = targetMap[promo.target] || promo.target
    96→    return `Diskon ${valueStr} (${targetStr})`
    97→  }
    98→
    99→  function parseMerchandise(data: string | null): Array<{ name: string; quantity: number }> {
   100→    if (!data) return []
   101→    try { return JSON.parse(data) } catch { return [] }
   102→  }
   103→
   104→  function parseSeatCodes(codes: string | null): string[] {
   105→    if (!codes) return []
   106→    try { return JSON.parse(codes) } catch {
   107→      return codes.split(',').map((s) => s.trim()).filter(Boolean)
   108→    }
   109→  }
   110→
   111→  function extractTransactionId(qrText: string): string | null {
   112→    const parts = qrText.split('KODE TRX:')
   113→    if (parts.length < 2) return null
   114→    return parts[1].trim()
   115→  }
   116→
   117→  async function performCheckIn(transactionId: string, overrideAction?: 'FORCE_VALID' | 'FORCE_INVALID' | 'RESET_COOLDOWN', overrideNote?: string) {
   118→    setScanState('LOADING')
   119→    setLastTransactionId(transactionId)
   120→    if (overrideAction) setOverrideLoading(overrideAction)
   121→
   122→    try {
   123→      const res = await fetch('/api/check-in', {
   124→        method: 'POST',
   125→        headers: { 'Content-Type': 'application/json' },
   126→        body: JSON.stringify({
   127→          transactionId,
   128→          overrideAction,
   129→          overrideNote: overrideNote || undefined,
   130→        }),
   131→      })
   132→
   133→      const data = await res.json()
   134→
   135→      if (data.status === 'SUCCESS') {
   136→        setScanState('SUCCESS')
   137→      } else if (data.status === 'WARNING') {
   138→        setScanState('WARNING')
   139→      } else {
   140→        setScanState('ERROR')
   141→      }
   142→
   143→      setResult(data)
   144→      // Track which day was targeted so the override UI can hint at it
   145→      setLastTargetDayId(data.transaction?.festival?.targetShowDateId || null)
   146→    } catch (err) {
   147→      setScanState('ERROR')
   148→      setResult({
   149→        status: 'ERROR',
   150→        message: 'Gagal terhubung ke server. Silakan coba lagi.',
   151→      })
   152→    } finally {
   153→      setOverrideLoading(null)
   154→    }
   155→  }
   156→
   157→  function handleScanSuccess(decodedText: string) {
   158→    const transactionId = extractTransactionId(decodedText)
   159→    if (!transactionId) {
   160→      setScanState('ERROR')
   161→      setResult({
   162→        status: 'ERROR',
   163→        message: 'Format QR tidak valid. Pastikan QR tiket yang benar.',
   164→      })
   165→      return
   166→    }
   167→    stopScanner()
   168→    performCheckIn(transactionId)
   169→  }
   170→
   171→  async function startScanner() {
   172→    await stopScanner()
   173→    try {
   174→      const { Html5Qrcode } = await import('html5-qrcode')
   175→      const scanner = new Html5Qrcode('qr-reader')
   176→      await scanner.start(
   177→        { facingMode: 'environment' },
   178→        {
   179→          fps: 10,
   180→          qrbox: { width: 280, height: 280 },
   181→          aspectRatio: 1.0,
   182→        },
   183→        (decodedText: string) => handleScanSuccess(decodedText),
   184→        () => { /* QR not found in frame - ignore */ }
   185→      )
   186→      scannerRef.current = scanner
   187→      setScanState('SCANNING')
   188→    } catch (err) {
   189→      console.error('Failed to start scanner:', err)
   190→      setScanState('ERROR')
   191→      setResult({
   192→        status: 'ERROR',
   193→        message: 'Gagal mengakses kamera. Pastikan izin kamera diberikan.',
   194→      })
   195→    }
   196→  }
   197→
   198→  async function stopScanner() {
   199→    if (scannerRef.current) {
   200→      try {
   201→        const state = scannerRef.current.getState()
   202→        if (state === 2) await scannerRef.current.stop()
   203→        scannerRef.current.clear()
   204→      } catch { /* ignore */ }
   205→      scannerRef.current = null
   206→    }
   207→  }
   208→
   209→  function resetScanner() {
   210→    setResult(null)
   211→    setLastTransactionId(null)
   212→    setLastTargetDayId(null)
   213→    setOverrideNote('')
   214→    setOverrideDialog(null)
   215→    setScanState('SCANNING')
   216→    setTimeout(() => startScanner(), 300)
   217→  }
   218→
   219→  function handleCheckInAgain() {
   220→    if (lastTransactionId) performCheckIn(lastTransactionId)
   221→  }

   538→                className="w-full min-h-[48px] bg-white hover:bg-white/90 rounded-xl"
   539→                style={{ color: '#D4843E' }}
   540→              >
   541→                {overrideLoading === 'RESET_COOLDOWN' ? (
   542→                  <Loader2 className="w-4 h-4 mr-2 animate-spin" />
   543→                ) : (
   544→                  <RefreshCw className="w-4 h-4 mr-2" />
   545→                )}
   546→                Reset Cooldown
   547→              </Button>
   548→            )}
   549→            <Button
   550→              onClick={() => openOverrideDialog('FORCE_VALID')}
   551→              disabled={overrideLoading === 'FORCE_VALID'}
   552→              variant="outline"
   553→              className="w-full min-h-[48px] bg-transparent border-white/40 text-white hover:bg-white/10 rounded-xl"
   554→            >
   555→              {overrideLoading === 'FORCE_VALID' ? (
   556→                <Loader2 className="w-4 h-4 mr-2 animate-spin" />
   557→              ) : (
   558→                <ShieldCheck className="w-4 h-4 mr-2" />
   559→              )}
   560→              Tetap Masukkan (Override)
   561→            </Button>
   562→            <Button
   563→              onClick={resetScanner}
   564→              className="w-full min-h-[48px] bg-white/10 hover:bg-white/20 text-white rounded-xl"
   565→            >
   566→              <QrCode className="w-4 h-4 mr-2" />
   567→              Scan Tiket Berikutnya
   568→            </Button>
   569→          </div>
   570→        </div>
   571→      </div>
   572→    )
   573→  }
   574→
   575→  // ── ERROR (wrong day, force-invalid, not found, etc.) ──
   576→  if (scanState === 'ERROR') {
   577→    const txn = result?.transaction
   578→    const allSeats = txn ? parseSeatCodes(txn.seatCodes) : []
   579→    const isFestival = allSeats.some((c) => c.includes('@'))
   580→    const festival = txn?.festival
   581→
   582→    return (
   583→      <div className="min-h-screen flex flex-col items-center justify-center p-6" style={{ backgroundColor: '#C75050' }}>
   584→        <div className="max-w-md w-full text-center text-white space-y-6">
   585→          {/* Icon */}
   586→          <div className="flex justify-center">
   587→            <div className="w-24 h-24 rounded-full bg-white/20 flex items-center justify-center">
   588→              <XCircle className="w-14 h-14 text-white" />
   589→            </div>
   590→          </div>
   591→
   592→          {/* Title */}
   593→          <div>
   594→            <h1 className="text-3xl font-bold font-serif">Tiket Tidak Valid</h1>
   595→            <p className="text-white/90 text-base font-medium mt-1">Tiket ditolak</p>
   596→          </div>
   597→
   598→          {/* Error Message */}
   599→          <div className="bg-white/15 rounded-xl p-5">
   600→            <p className="text-white/90 text-base">{result?.message || 'Terjadi kesalahan saat memverifikasi tiket.'}</p>
   601→          </div>
   602→
   603→          {/* If festival wrong-day, show which days ARE valid */}
   604→          {isFestival && festival && (
   605→            <div className="bg-white/10 rounded-xl p-4 text-left">
   606→              <p className="text-white/70 text-xs uppercase tracking-wider mb-2">Hari Berlaku di Paket Ini:</p>
   607→              <div className="space-y-1.5">
   608→                {festival.applicableShowDates.map((d) => {
   609→                  const date = new Date(d.date)
   610→                  const weekday = date.toLocaleDateString('id-ID', { weekday: 'short' })
   611→                  const dayMonth = date.toLocaleDateString('id-ID', { day: 'numeric', month: 'short' })
   612→                  const time = `${date.toLocaleTimeString('id-ID', { hour: '2-digit', minute: '2-digit' })} WIB`
   613→                  return (
   614→                    <div key={d.id} className="flex items-center justify-between text-sm">
   615→                      <span className="font-mono">{weekday}, {dayMonth} · {time}</span>
   616→                      {d.isScanned && <CheckCircle className="w-3.5 h-3.5 text-white/70" />}
   617→                    </div>
   618→                  )
   619→                })}
   620→              </div>
   621→            </div>
   622→          )}
   623→
   624→          {/* Customer Info (if we have it) */}
   625→          {txn && (
   626→            <div className="bg-white/15 rounded-xl p-4 text-left text-white">
   627→              <p className="text-white/60 text-xs uppercase tracking-wider">Nama Pemesan</p>
   628→              <p className="text-base font-semibold">{txn.customerName}</p>
   629→              <p className="text-white/60 text-xs mt-2">TRX: {txn.transactionId}</p>
   630→            </div>
   631→          )}
   632→
   633→          {/* Override Action (only if we have a transaction) */}
   634→          {txn && (
   635→            <Button
   636→              onClick={() => openOverrideDialog('FORCE_VALID')}
   637→              disabled={overrideLoading === 'FORCE_VALID'}
   638→              variant="outline"
   639→              className="w-full min-h-[48px] bg-transparent border-white/40 text-white hover:bg-white/10 rounded-xl"
   640→            >
   641→              {overrideLoading === 'FORCE_VALID' ? (
   642→                <Loader2 className="w-4 h-4 mr-2 animate-spin" />
   643→              ) : (
   644→                <ShieldCheck className="w-4 h-4 mr-2" />
   645→              )}
   646→              Tetap Masukkan (Override)
   647→            </Button>
   648→          )}
   649→
   650→          <Button
   651→            onClick={resetScanner}
   652→            className="w-full min-h-[56px] text-lg font-semibold bg-white hover:bg-white/90 rounded-xl"
   653→            style={{ color: '#C75050' }}
   654→          >
   655→            <RotateCcw className="w-5 h-5 mr-2" />
   656→            Scan Ulang
   657→          </Button>
   658→        </div>
   659→      </div>
   660→    )
   661→  }
   662→
   663→  // ── Override Note Dialog ──
   664→  const overrideTitle =
   665→    overrideDialog === 'FORCE_VALID' ? 'Override: Tetap Masukkan'
   666→    : overrideDialog === 'FORCE_INVALID' ? 'Override: Tolak Tiket'
   667→    : overrideDialog === 'RESET_COOLDOWN' ? 'Reset Cooldown'
   668→    : ''
   669→  const overrideDesc =
   670→    overrideDialog === 'FORCE_VALID' ? 'Tiket akan diizinkan masuk meskipun tidak valid. Catat alasannya untuk audit.'
   671→    : overrideDialog === 'FORCE_INVALID' ? 'Tiket akan ditolak secara permanen. Catat alasannya untuk audit.'
   672→    : overrideDialog === 'RESET_COOLDOWN' ? 'Cooldown untuk hari ini akan direset. Penonton bisa langsung di-scan ulang.'
   673→    : ''
   674→  const overrideIcon =
   675→    overrideDialog === 'FORCE_VALID' ? ShieldCheck
   676→    : overrideDialog === 'FORCE_INVALID' ? ShieldX
   677→    : RefreshCw
   678→
   679→  const OverrideIcon = overrideIcon
   680→
   681→  return (
   682→    <div className="min-h-screen flex flex-col bg-warm-white">
   683→      {/* Header */}
   684→      <div className="bg-charcoal text-white px-4 py-4 flex items-center gap-3">
   685→        <Button
   686→          variant="ghost"
   687→          size="icon"
   688→          className="text-white/70 hover:text-white hover:bg-white/10 h-11 w-11"
   689→          onClick={async () => {
   690→            await stopScanner()
   691→            window.history.back()
   692→          }}
   693→        >
   694→          <ArrowLeft className="w-5 h-5" />
   695→        </Button>
   696→        <div>
   697→          <h1 className="font-serif text-lg font-semibold">Scanner Tiket</h1>
   698→          <p className="text-white/50 text-xs">Arahkan QR Code ke kamera</p>
   699→        </div>
   700→      </div>
   701→
   702→      {/* Scanner Area */}
   703→      <div className="flex-1 flex flex-col items-center justify-center p-4">
   704→        <div className="w-full max-w-sm">
   705→          {/* Scanner Container */}
   706→          <div
   707→            ref={scannerContainerRef}
   708→            id="qr-reader"
   709→            className="rounded-2xl overflow-hidden shadow-lg border-2 border-charcoal/20"
   710→            style={{ minHeight: '320px' }}
   711→          />
   712→
   713→          {/* Instructions */}
   714→          <div className="mt-6 text-center space-y-2">
   715→            <div className="flex items-center justify-center gap-2 text-charcoal/70">
   716→              <QrCode className="w-5 h-5" />
   717→              <p className="text-sm font-medium">Menunggu QR Code...</p>
   718→            </div>
   719→            <p className="text-xs text-muted-foreground">
   720→              Pastikan QR Code berada dalam bingkai scanner
   721→            </p>
   722→          </div>
   723→        </div>
   724→      </div>
   725→
   726→      {/* Footer Info */}
   727→      <div className="px-4 py-4 border-t border-border/50">
   728→        <p className="text-center text-xs text-muted-foreground">
   729→          QR Format: NAMA: ... | KURSI/FESTIVAL: ... | KODE TRX: ...
   730→        </p>
   731→      </div>
   732→
   733→      {/* Override Note Dialog */}
   734→      <Dialog open={overrideDialog !== null} onOpenChange={(open) => { if (!open) setOverrideDialog(null) }}>
   735→        <DialogContent className="max-w-md">
   736→          <DialogHeader>
   737→            <DialogTitle className="font-serif text-lg flex items-center gap-2">
   738→              {OverrideIcon && <OverrideIcon className="w-5 h-5 text-gold" />}
   739→              {overrideTitle}
   740→            </DialogTitle>
   741→            <DialogDescription>{overrideDesc}</DialogDescription>
   742→          </DialogHeader>
   743→
   744→          <div className="space-y-3">
   745→            <Label htmlFor="override-note" className="text-sm font-medium">
   746→              Alasan Override (wajib untuk audit log)
   747→            </Label>
   748→            <Input
   749→              id="override-note"
   750→              placeholder="Contoh: Penonton keluar toilet sejenak, sistem cooldown error"
   751→              value={overrideNote}
   752→              onChange={(e) => setOverrideNote(e.target.value)}
   753→              autoFocus
   754→            />
   755→            {!overrideNote.trim() && (
   756→              <p className="text-[11px] text-muted-foreground">
   757→                Catatan ini akan tersimpan di log scan tiket.
   758→              </p>
   759→            )}
   760→          </div>
   761→
   762→          <DialogFooter className="gap-2">
   763→            <Button variant="outline" onClick={() => setOverrideDialog(null)}>
   764→              Batal
   765→            </Button>
   766→            <Button
   767→              onClick={confirmOverride}
   768→              disabled={!overrideNote.trim() || overrideLoading !== null}
   769→              className={
   770→                overrideDialog === 'FORCE_INVALID'
   771→                  ? 'bg-danger hover:bg-danger/90 text-white'
   772→                  : 'bg-charcoal hover:bg-charcoal/90 text-gold'
   773→              }
   774→            >
   775→              {overrideLoading !== null ? (
   776→                <Loader2 className="w-4 h-4 mr-2 animate-spin" />
   777→              ) : overrideDialog === 'FORCE_VALID' ? (
   778→                <ShieldCheck className="w-4 h-4 mr-2" />
   779→              ) : overrideDialog === 'FORCE_INVALID' ? (
   780→                <ShieldX className="w-4 h-4 mr-2" />
   781→              ) : (
   782→                <RefreshCw className="w-4 h-4 mr-2" />
   783→              )}
   784→              Konfirmasi Override
   785→            </Button>
