   330→  async function handleSave() {
   331→    setIsSaving(true)
   332→    try {
   333→      const url = editingId ? `/api/admin/events/${editingId}` : '/api/admin/events'
   334→      const method = editingId ? 'PUT' : 'POST'
   335→
   336→      const isFestival = formData.eventMode === 'FESTIVAL'
   337→
   338→      // Build a map of showDate tempId → existing DB id (for editing existing events)
   339→      // For new showDates (no DB id), they'll be saved first then matched to price categories in a second pass
   340→      // Use Record (plain object) — avoids conflict with lucide-react's `Map` icon import
   341→      const showDateTempToDbId: Record<string, string> = {}
   342→      formData.showDates.forEach(sd => {
   343→        if (sd.id && sd.tempId) showDateTempToDbId[sd.tempId] = sd.id
   344→      })
   345→
   346→      // Festival Mode: filter out empty price categories and serialize applicableDayIds
   347→      const serializedPriceCategories = formData.priceCategories
   348→        .filter(pc => pc.name.trim())
   349→        .map(pc => {
   350→          if (!isFestival) {
   351→            // Regular event: no packageType / applicableDayIds
   352→            const { packageType, applicableDayIds, ...rest } = pc
   353→            void packageType; void applicableDayIds
   354→            return rest
   355→          }
   356→          // Festival Mode
   357→          const pkgType = pc.packageType || 'SINGLE'
   358→          // FULL = all days (null), SINGLE/MULTI = specific days
   359→          if (pkgType === 'FULL') {
   360→            const { packageType: _pt, applicableDayIds: _ad, ...rest } = pc
   361→            void _pt; void _ad
   362→            return { ...rest, packageType: 'FULL', applicableDayIds: null }
   363→          }
   364→          // SINGLE or MULTI: convert tempIds to DB IDs where possible (fallback to tempId for new showDates)
   365→          const dayIds = (pc.applicableDayIds || []).map(tempId =>
   366→            showDateTempToDbId[tempId] || tempId
   367→          )
   368→          return {
   369→            name: pc.name,
   370→            price: pc.price,
   371→            colorCode: pc.colorCode,
   372→            packageType: pkgType,
   373→            applicableDayIds: JSON.stringify(dayIds),
   374→          }
   375→        })
   376→
   377→      // ─── GA Integration: auto-build gaZoneConfig from price categories ──
   378→      // For GA events (REGULAR+GA or FESTIVAL), auto-generate one zone per
   379→      // price category so admin doesn't have to redefine zones in seats page.
   380→      const effectiveSeatType = isFestival ? 'GENERAL_ADMISSION' : formData.seatType
   381→      const isGA = effectiveSeatType === 'GENERAL_ADMISSION'
   382→
   383→      // Preserve existing zone notes if a zone with the same name already exists
   384→      let existingZones: Array<{ name?: string; priceCategoryName?: string; notes?: string }> = []
   385→      try {
   386→        if (formData.gaZoneConfig) {
   387→          existingZones = JSON.parse(formData.gaZoneConfig)
   388→        }
   389→      } catch { /* ignore */ }
   390→
   391→      // Build new gaZoneConfig from price categories (1 zone per category)
   392→      const gaZoneConfig = isGA
   393→        ? JSON.stringify(formData.priceCategories
   394→            .filter(pc => pc.name.trim())
   395→            .map(pc => {
   396→              const match = existingZones.find(z =>
   397→                z.priceCategoryName === pc.name || z.name === pc.name)
   398→              return {
   399→                name: pc.name,
   400→                capacity: pc.capacity ?? 100,
   401→                price: pc.price,
   402→                color: pc.colorCode,
   403→                priceCategoryName: pc.name,
   404→                notes: match?.notes || '',
   405→              }
   406→            }))
   407→        : undefined
   408→
   409→      const payload = {
   410→        ...formData,
   411→        // Use first showDate entry as the primary showDate (backward compat)
   412→        showDate: formData.showDates[0]?.date
   413→          ? new Date(formData.showDates[0].date).toISOString()
   414→          : new Date().toISOString(),
   415→        openGate: formData.showDates[0]?.openGate
   416→          ? new Date(formData.showDates[0].openGate).toISOString()
   417→          : null,
   418→        showDates: formData.showDates
   419→          .filter((sd) => sd.date)
   420→          .map(sd => ({ id: sd.id, date: sd.date, openGate: sd.openGate, label: sd.label })),
   421→        priceCategories: serializedPriceCategories,
   422→        // Festival Mode
   423→        eventMode: formData.eventMode,
   424→        scanCooldownMinutes: formData.scanCooldownMinutes,
   425→        cooldownEnabled: formData.cooldownEnabled,
   426→        // Force seatType = GA when FESTIVAL
   427→        seatType: isFestival ? 'GENERAL_ADMISSION' : formData.seatType,
   428→        // GA Integration: auto-sync gaZoneConfig from price categories
   429→        ...(gaZoneConfig !== undefined ? { gaZoneConfig } : {}),
   430→      }
   431→
   432→      const res = await fetch(url, {
   433→        method,
   434→        headers: { 'Content-Type': 'application/json' },
   435→        body: JSON.stringify(payload),
   436→      })
   437→
   438→      if (res.ok) {
   439→        // For FESTIVAL events with new showDates, we need a second pass to update applicableDayIds
   440→        // with real DB IDs (since new showDates only got their IDs after the first save)
   441→        if (isFestival && editingId) {
   442→          // Refetch event to get actual DB IDs for showDates
   443→          const refetchRes = await fetch(`/api/admin/events/${editingId}`)
   444→          if (refetchRes.ok) {
   445→            const refetchData = await refetchRes.json()
   446→            const updatedShowDates = refetchData.event.showDates || []
   447→            // Build new tempId → DB ID map using position in array (since order is preserved by date asc)
   448→            const newTempToDbMap: Record<string, string> = {}
   449→            const sortedShowDates = [...formData.showDates].filter(sd => sd.date)
   450→              .sort((a, b) => new Date(a.date).getTime() - new Date(b.date).getTime())
   451→            updatedShowDates.forEach((dbSd: { id: string }, idx: number) => {
   452→              if (sortedShowDates[idx]?.tempId) {
   453→                newTempToDbMap[sortedShowDates[idx].tempId!] = dbSd.id
   454→              }
   455→            })
   456→            // Check if any price category has applicableDayIds that need updating
   457→            const needsUpdate = formData.priceCategories.some(pc =>
   458→              pc.packageType && pc.packageType !== 'FULL' &&
   459→              (pc.applicableDayIds || []).some(tempId => !showDateTempToDbId[tempId])
   460→            )
   461→            if (needsUpdate) {
   462→              const updatedPriceCats = formData.priceCategories
   463→                .filter(pc => pc.name.trim())
   464→                .map(pc => {
   465→                  if (!pc.packageType || pc.packageType === 'FULL') {
   466→                    return {
   467→                      name: pc.name,
   468→                      price: pc.price,
   469→                      colorCode: pc.colorCode,
   470→                      packageType: pc.packageType || null,
   471→                      applicableDayIds: null,
   472→                    }
   473→                  }
   474→                  const dayIds = (pc.applicableDayIds || []).map(tempId =>
   475→                    newTempToDbMap[tempId] || showDateTempToDbId[tempId] || tempId
   476→                  )
   477→                  return {
   478→                    name: pc.name,
   479→                    price: pc.price,
   480→                    colorCode: pc.colorCode,
   481→                    packageType: pc.packageType,
   482→                    applicableDayIds: JSON.stringify(dayIds),
   483→                  }
   484→                })
   485→              // Send second update to fix applicableDayIds
   486→              await fetch(`/api/admin/events/${editingId}`, {
   487→                method: 'PUT',
   488→                headers: { 'Content-Type': 'application/json' },
   489→                body: JSON.stringify({ priceCategories: updatedPriceCats }),
   490→              })
   491→            }
   492→          }
   493→        }
   494→        setIsDialogOpen(false)
   495→        fetchEvents()
   496→      } else {
   497→        const data = await res.json()
   498→        alert(data.error || 'Gagal menyimpan event')
   499→      }
   500→    } catch (err) {
   501→      console.error('Save error:', err)
   502→      alert('Terjadi kesalahan')
   503→    } finally {
   504→      setIsSaving(false)
   505→    }
   506→  }
