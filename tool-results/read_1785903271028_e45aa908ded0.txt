   385→  async function handleSubmit() {
   386→    if (!selectedEventId || !guestName || !guestEmail || !guestPhone) {
   387→      setSubmitResult({ success: false, message: 'Harap lengkapi semua field yang wajib diisi.' })
   388→      return
   389→    }
   390→
   391→    let seatCodes: string[]
   392→    let festivalPackage: { priceCategoryId: string; quantity: number } | undefined
   393→
   394→    if (isFestival) {
   395→      // ─── FESTIVAL MODE: pick package + quantity ────────────────────
   396→      if (!selectedFestivalPkg) {
   397→        setSubmitResult({ success: false, message: 'Pilih paket festival terlebih dahulu.' })
   398→        return
   399→      }
   400→      if (festivalQuantity < 1) {
   401→        setSubmitResult({ success: false, message: 'Jumlah tiket minimal 1.' })
   402→        return
   403→      }
   404→      const avail = festivalAvailability[selectedFestivalPkg.id] ?? 0
   405→      if (festivalQuantity > avail) {
   406→        setSubmitResult({
   407→          success: false,
   408→          message: `Tiket tersedia hanya ${avail}. Kurangi jumlah atau pilih paket lain.`,
   409→        })
   410→        return
   411→      }
   412→      // Seat codes: use package name as prefix (matches zoneName in seat table)
   413→      seatCodes = Array.from({ length: festivalQuantity }, (_, i) =>
   414→        `${selectedFestivalPkg.name}-COMP${i + 1}`
   415→      )
   416→      festivalPackage = {
   417→        priceCategoryId: selectedFestivalPkg.id,
   418→        quantity: festivalQuantity,
   419→      }
   420→    } else if (isNumberedSeatMap) {
   421→      if (selectedSeats.length === 0) {
   422→        setSubmitResult({ success: false, message: 'Harap pilih minimal 1 kursi.' })
   423→        return
   424→      }
   425→      seatCodes = selectedSeats
   426→    } else {
   427→      if (gaQuantity < 1) {
   428→        setSubmitResult({ success: false, message: 'Jumlah tiket minimal 1.' })
   429→        return
   430→      }
   431→      // Generate GA seat codes
   432→      seatCodes = Array.from({ length: gaQuantity }, (_, i) =>
   433→        gaZone ? `${gaZone}-${i + 1}` : `GA-${i + 1}`
   434→      )
   435→    }
   436→
   437→    setIsSubmitting(true)
   438→    setSubmitResult(null)
   439→
   440→    try {
   441→      const res = await fetch('/api/admin/tickets/complimentary', {
   442→        method: 'POST',
   443→        headers: { 'Content-Type': 'application/json' },
   444→        body: JSON.stringify({
   445→          eventId: selectedEventId,
   446→          seatCodes,
   447→          guestName,
   448→          guestEmail,
   449→          guestPhone,
   450→          showDateId: selectedShowDateId || undefined,
   451→          festivalPackage,
   452→        }),
   453→      })
   454→
   455→      const data = await res.json()
   456→
   457→      if (res.ok) {
   458→        setSubmitResult({
   459→          success: true,
   460→          message: `Tiket komplimen berhasil dibuat! Transaction ID: ${data.transactionId}`,
   461→        })
   462→        // Reset form
   463→        setGuestName('')
   464→        setGuestEmail('')
   465→        setGuestPhone('')
   466→        setSelectedSeats([])
   467→        setGaQuantity(1)
   468→        setGaZone('')
   469→        setFestivalQuantity(1)
   470→        setSelectedFestivalPkgId('')
   471→        // Refresh recent tickets
   472→        const recentRes = await fetch('/api/admin/tickets/complimentary')
   473→        if (recentRes.ok) {
   474→          const recentData = await recentRes.json()
   475→          setRecentTickets(recentData.tickets || [])
   476→          setTotalComplimentary(recentData.total || 0)
   477→        }
   478→      } else {
   479→        setSubmitResult({
   480→          success: false,
   481→          message: data.error || 'Gagal membuat tiket komplimen.',
   482→        })
   483→      }
   484→    } catch (err) {
   485→      setSubmitResult({
   486→        success: false,
   487→        message: 'Terjadi kesalahan jaringan. Silakan coba lagi.',
   488→      })
   489→    } finally {
   490→      setIsSubmitting(false)
   491→    }
   492→  }
