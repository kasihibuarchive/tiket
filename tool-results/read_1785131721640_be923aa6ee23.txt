   197→   197→function safeParseArray<T>(raw: string | null | undefined, fallback: T[]): T[] {
   198→   198→  if (!raw) return fallback
   199→   199→  try {
   200→   200→    const parsed = JSON.parse(raw)
   201→   201→    return Array.isArray(parsed) ? parsed : fallback
   202→   202→  } catch {
   203→   203→    return fallback
   204→   204→  }
