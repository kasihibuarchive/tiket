   196→// ─── Helper: safe JSON parse ──────────────────────────────────────

   198→function safeParseArray<T>(raw: string | null | undefined, fallback: T[]): T[] {
   199→  if (!raw) return fallback
   200→  try {
   201→    const parsed = JSON.parse(raw)
   202→    return Array.isArray(parsed) ? parsed : fallback
   203→  } catch {
   204→    return fallback
   205→  }
   206→}

   260→function ShortLinksCard({ eventId }: { eventId: string }) {
   261→  const [shortLinks, setShortLinks] = useState<ShortLinkData[]>([])
   262→  const [isLoading, setIsLoading] = useState(true)
   263→  const [newSlug, setNewSlug] = useState('')
   264→  const [isCreating, setIsCreating] = useState(false)
   265→  const [error, setError] = useState<string | null>(null)
   266→  const [copiedSlug, setCopiedSlug] = useState<string | null>(null)
   267→
   268→  useEffect(() => {
   269→    fetchShortLinks()
   270→  }, [eventId])
   271→
   272→  async function fetchShortLinks() {
   273→    try {
   274→      const res = await fetch(`/api/admin/short-links?eventId=${eventId}`)
   275→      if (res.ok) {
   276→        const data = await res.json()
   277→        setShortLinks(data.shortLinks || [])
   278→      }
   279→    } catch (err) {
   280→      console.error('Fetch short links error:', err)
   281→    } finally {
   282→      setIsLoading(false)
   283→    }
   284→  }
   285→
   286→  async function handleCreate() {
   287→    if (!newSlug.trim()) return
   288→    setIsCreating(true)
   289→    setError(null)
   290→    try {
   291→      const res = await fetch('/api/admin/short-links', {
   292→        method: 'POST',
   293→        headers: { 'Content-Type': 'application/json' },
   294→        body: JSON.stringify({ slug: newSlug.trim(), eventId }),
   295→      })
   296→      if (res.ok) {
   297→        setNewSlug('')
   298→        fetchShortLinks()
   299→      } else {
   300→        const data = await res.json()
   301→        setError(data.error || 'Gagal membuat short link')
   302→      }
   303→    } catch {
   304→      setError('Terjadi kesalahan')
   305→    } finally {
   306→      setIsCreating(false)
   307→    }
   308→  }
   309→
   310→  async function handleDelete(id: string, slug: string) {
   311→    if (!confirm(`Hapus short link "${slug}"?`)) return
   312→    try {
   313→      const res = await fetch(`/api/admin/short-links/${id}`, { method: 'DELETE' })
   314→      if (res.ok) {
   315→        setShortLinks(shortLinks.filter((sl) => sl.id !== id))
   316→      }
   317→    } catch (err) {
   318→      console.error('Delete short link error:', err)
   319→    }
   320→  }
   321→
   322→  function handleCopy(slug: string) {
   323→    const url = `${window.location.origin}/${slug}`
   324→    navigator.clipboard.writeText(url)
   325→    setCopiedSlug(slug)
   326→    setTimeout(() => setCopiedSlug(null), 2000)
   327→  }
   328→
   329→  return (
   330→    <Card className="border-border/50">
   331→      <CardHeader className="pb-3">
   332→        <div className="flex items-center gap-2">
   333→          <Link2 className="w-5 h-5 text-gold" />
   334→          <CardTitle className="font-serif text-lg text-charcoal">Short Links</CardTitle>
   335→          <Badge variant="secondary" className="text-xs bg-gold/10 text-gold-dark">
   336→            {shortLinks.length} link
   337→          </Badge>
   338→        </div>
   339→      </CardHeader>
   340→      <CardContent className="space-y-4">
   341→        {/* Create new short link */}
   342→        <div className="border border-dashed border-border/60 rounded-lg p-4">
   343→          <p className="text-xs font-medium text-muted-foreground mb-3">Buat Short Link Baru</p>
   344→          <div className="flex items-center gap-2">
   345→            <div className="flex items-center bg-muted/50 border border-border/60 rounded-md px-3 h-9 text-sm text-muted-foreground shrink-0">
   346→              teateran.site/
   347→            </div>
   348→            <Input
   349→              value={newSlug}
   350→              onChange={(e) => {
   351→                setNewSlug(e.target.value.replace(/[^a-zA-Z0-9_-]/g, '').slice(0, 30))
   352→                setError(null)
   353→              }}
   354→              placeholder="tajoc"
   355→              className="h-9"
   356→              onKeyDown={(e) => e.key === 'Enter' && handleCreate()}
   357→            />
   358→            <Button
   359→              onClick={handleCreate}
   360→              disabled={isCreating || !newSlug.trim()}
   361→              size="sm"
   362→              className="bg-gold hover:bg-gold/90 text-charcoal shrink-0"
   363→            >
   364→              {isCreating ? <Loader2 className="w-3.5 h-3.5 animate-spin" /> : <Plus className="w-3.5 h-3.5 mr-1" />}
   365→              Buat
   366→            </Button>
   367→          </div>
   368→          {error && <p className="text-xs text-red-500 mt-2">{error}</p>}
   369→          <p className="text-xs text-muted-foreground/60 mt-2">
   370→            Hanya huruf, angka, strip (-), underscore (_). 2-30 karakter. Sekali pakai per event.
   371→          </p>
   372→        </div>
   373→
   374→        {/* List of existing short links */}
   375→        {isLoading ? (
   376→          <div className="flex justify-center py-4">
   377→            <Loader2 className="w-5 h-5 text-gold animate-spin" />
   378→          </div>
   379→        ) : shortLinks.length === 0 ? (
   380→          <div className="text-center py-6 rounded-lg border border-dashed border-border/60 bg-muted/20">
   381→            <Link2 className="w-8 h-8 text-muted-foreground/30 mx-auto mb-2" />
   382→            <p className="text-sm text-muted-foreground">Belum ada short link</p>
   383→            <p className="text-xs text-muted-foreground/60">Buat short link di atas untuk mempermudah berbagi</p>
   384→          </div>
   385→        ) : (
   386→          <div className="space-y-2 max-h-64 overflow-y-auto">
   387→            {shortLinks.map((sl) => (
   388→              <div
   389→                key={sl.id}
   390→                className="rounded-lg border border-border/60 p-3 bg-white flex items-center gap-3"
   391→              >
   392→                <div className="flex-1 min-w-0">
   393→                  <div className="flex items-center gap-2">

   405→                  <Button
