    31→function GuestReviewsCard({ eventId, isCompleted }: { eventId: string; isCompleted: boolean }) {

    54→      }
    55→    } catch (err) {
    56→      console.error('Delete review error:', err)
    57→    }
    58→  }
    59→
    60→  if (!isCompleted) return null
    61→
    62→  return (
    63→    <Card className="border-border/50">
    64→      <CardHeader className="pb-3">
    65→        <div className="flex items-center justify-between">
    66→          <div className="flex items-center gap-2">
    67→            <MessageSquareQuote className="w-5 h-5 text-gold" />
    68→            <CardTitle className="font-serif text-lg text-charcoal">Guest Reviews</CardTitle>

    70→              {reviews.length} review
    71→            </Badge>

    74→            <div className="flex items-center gap-1">
    75→              <Star className="w-4 h-4 text-gold fill-gold" />
    76→              <span className="text-sm font-semibold text-charcoal">{stats.average}</span>
    77→              <span className="text-xs text-muted-foreground">/5</span>
    78→            </div>
    79→          )}
    80→        </div>
    81→      </CardHeader>
    82→      <CardContent>
    83→        {isLoading ? (
    84→          <div className="flex justify-center py-6">
    85→            <Loader2 className="w-5 h-5 text-gold animate-spin" />
    86→          </div>
    87→        ) : reviews.length === 0 ? (
    88→          <div className="text-center py-6 rounded-lg border border-dashed border-border/60 bg-muted/20">
    89→            <MessageSquareQuote className="w-8 h-8 text-muted-foreground/30 mx-auto mb-2" />

    91→          </div>
    92→        ) : (
    93→          <div className="space-y-2 max-h-96 overflow-y-auto">
    94→            {reviews.map((review) => (
    95→              <div
    96→                key={review.id}
    97→                className="rounded-lg border border-border/60 p-3 bg-white"
    98→              >
    99→                <div className="flex items-center justify-between">
   100→                  <div className="flex items-center gap-2">
   101→                    <span className="text-sm font-medium text-charcoal">{review.authorName}</span>

   195→// ─── Helper: safe JSON parse ──────────────────────────────────────
   196→
   197→function safeParseArray<T>(raw: string | null | undefined, fallback: T[]): T[] {
   198→  if (!raw) return fallback
   199→  try {
   200→    const parsed = JSON.parse(raw)
   201→    return Array.isArray(parsed) ? parsed : fallback
   202→  } catch {
   203→    return fallback
   204→  }
   205→}

   259→function ShortLinksCard({ eventId }: { eventId: string }) {
   260→  const [shortLinks, setShortLinks] = useState<ShortLinkData[]>([])
   261→  const [isLoading, setIsLoading] = useState(true)
   262→  const [newSlug, setNewSlug] = useState('')
   263→  const [isCreating, setIsCreating] = useState(false)
   264→  const [error, setError] = useState<string | null>(null)
   265→  const [copiedSlug, setCopiedSlug] = useState<string | null>(null)
   266→
   267→  useEffect(() => {
   268→    fetchShortLinks()
   269→  }, [eventId])
   270→
   271→  async function fetchShortLinks() {
   272→    try {
   273→      const res = await fetch(`/api/admin/short-links?eventId=${eventId}`)
   274→      if (res.ok) {
   275→        const data = await res.json()
   276→        setShortLinks(data.shortLinks || [])
   277→      }
   278→    } catch (err) {
   279→      console.error('Fetch short links error:', err)
   280→    } finally {
   281→      setIsLoading(false)
   282→    }
   283→  }
   284→
   285→  async function handleCreate() {
   286→    if (!newSlug.trim()) return
   287→    setIsCreating(true)
   288→    setError(null)
   289→    try {
   290→      const res = await fetch('/api/admin/short-links', {
   291→        method: 'POST',
   292→        headers: { 'Content-Type': 'application/json' },
   293→        body: JSON.stringify({ slug: newSlug.trim(), eventId }),
   294→      })
   295→      if (res.ok) {
   296→        setNewSlug('')
   297→        fetchShortLinks()
   298→      } else {
   299→        const data = await res.json()
   300→        setError(data.error || 'Gagal membuat short link')
   301→      }
   302→    } catch {
   303→      setError('Terjadi kesalahan')
   304→    } finally {
   305→      setIsCreating(false)
   306→    }
   307→  }
   308→
   309→  async function handleDelete(id: string, slug: string) {
   310→    if (!confirm(`Hapus short link "${slug}"?`)) return
   311→    try {
   312→      const res = await fetch(`/api/admin/short-links/${id}`, { method: 'DELETE' })
   313→      if (res.ok) {
   314→        setShortLinks(shortLinks.filter((sl) => sl.id !== id))
   315→      }
   316→    } catch (err) {
   317→      console.error('Delete short link error:', err)
   318→    }
   319→  }
   320→
   321→  function handleCopy(slug: string) {
   322→    const url = `${window.location.origin}/${slug}`
   323→    navigator.clipboard.writeText(url)
   324→    setCopiedSlug(slug)
   325→    setTimeout(() => setCopiedSlug(null), 2000)
   326→  }
   327→
   328→  return (
   329→    <Card className="border-border/50">
   330→      <CardHeader className="pb-3">
   331→        <div className="flex items-center gap-2">
   332→          <Link2 className="w-5 h-5 text-gold" />
   333→          <CardTitle className="font-serif text-lg text-charcoal">Short Links</CardTitle>
   334→          <Badge variant="secondary" className="text-xs bg-gold/10 text-gold-dark">
   335→            {shortLinks.length} link
   336→          </Badge>
   337→        </div>
   338→      </CardHeader>
   339→      <CardContent className="space-y-4">
   340→        {/* Create new short link */}
   341→        <div className="border border-dashed border-border/60 rounded-lg p-4">
   342→          <p className="text-xs font-medium text-muted-foreground mb-3">Buat Short Link Baru</p>
   343→          <div className="flex items-center gap-2">
   344→            <div className="flex items-center bg-muted/50 border border-border/60 rounded-md px-3 h-9 text-sm text-muted-foreground shrink-0">
   345→              teateran.site/
   346→            </div>
   347→            <Input
   348→              value={newSlug}
   349→              onChange={(e) => {
   350→                setNewSlug(e.target.value.replace(/[^a-zA-Z0-9_-]/g, '').slice(0, 30))
   351→                setError(null)
   352→              }}
   353→              placeholder="tajoc"
   354→              className="h-9"
   355→              onKeyDown={(e) => e.key === 'Enter' && handleCreate()}
   356→            />
   357→            <Button
   358→              onClick={handleCreate}
   359→              disabled={isCreating || !newSlug.trim()}
   360→              size="sm"
   361→              className="bg-gold hover:bg-gold/90 text-charcoal shrink-0"
   362→            >
   363→              {isCreating ? <Loader2 className="w-3.5 h-3.5 animate-spin" /> : <Plus className="w-3.5 h-3.5 mr-1" />}
   364→              Buat
   365→            </Button>
   366→          </div>
   367→          {error && <p className="text-xs text-red-500 mt-2">{error}</p>}
   368→          <p className="text-xs text-muted-foreground/60 mt-2">
   369→            Hanya huruf, angka, strip (-), underscore (_). 2-30 karakter. Sekali pakai per event.
   370→          </p>
   371→        </div>
   372→
   373→        {/* List of existing short links */}
   374→        {isLoading ? (
   375→          <div className="flex justify-center py-4">
   376→            <Loader2 className="w-5 h-5 text-gold animate-spin" />
   377→          </div>
   378→        ) : shortLinks.length === 0 ? (
   379→          <div className="text-center py-6 rounded-lg border border-dashed border-border/60 bg-muted/20">
   380→            <Link2 className="w-8 h-8 text-muted-foreground/30 mx-auto mb-2" />
   381→            <p className="text-sm text-muted-foreground">Belum ada short link</p>
   382→            <p className="text-xs text-muted-foreground/60">Buat short link di atas untuk mempermudah berbagi</p>
   383→          </div>
   384→        ) : (
   385→          <div className="space-y-2 max-h-64 overflow-y-auto">
   386→            {shortLinks.map((sl) => (
   387→              <div
   388→                key={sl.id}
   389→                className="rounded-lg border border-border/60 p-3 bg-white flex items-center gap-3"
   390→              >
   391→                <div className="flex-1 min-w-0">
   392→                  <div className="flex items-center gap-2">
   393→                    <span className="text-sm font-medium text-charcoal">/{sl.slug}</span>
   394→                    <Badge variant="secondary" className="text-[10px] bg-gold/10 text-gold-dark px-1.5 py-0">
   395→                      <MousePointerClick className="w-2.5 h-2.5 mr-0.5" />
   396→                      {sl.clickCount}
   397→                    </Badge>
   398→                  </div>
   399→                  <p className="text-xs text-muted-foreground/60 mt-0.5">
   400→                    teateran.site/{sl.slug}
   401→                  </p>
   402→                </div>
   403→                <div className="flex items-center gap-1 shrink-0">
   404→                  <Button
   405→                    variant="ghost"
   406→                    size="icon"
   407→                    className="h-7 w-7 text-muted-foreground hover:text-charcoal"
   408→                    onClick={() => handleCopy(sl.slug)}
   409→                    title="Copy link"
   410→                  >
   411→                    {copiedSlug === sl.slug ? <Check className="w-3.5 h-3.5 text-emerald-500" /> : <Copy className="w-3.5 h-3.5" />}
   412→                  </Button>
   413→                  <a
   414→                    href={`/${sl.slug}`}
   415→                    target="_blank"
   416→                    rel="noopener noreferrer"
   417→                    className="h-7 w-7 inline-flex items-center justify-center rounded-md text-muted-foreground hover:text-charcoal hover:bg-accent"
   418→                    title="Buka link"
   419→                  >
   420→                    <ExternalLink className="w-3.5 h-3.5" />
   421→                  </a>
   422→                  <Button
   423→                    variant="ghost"
   424→                    size="icon"
   425→                    className="h-7 w-7 text-red-400 hover:text-red-600 hover:bg-red-50"
   426→                    onClick={() => handleDelete(sl.id, sl.slug)}
   427→                    title="Hapus"
   428→                  >
   429→                    <Trash2 className="w-3.5 h-3.5" />
   430→                  </Button>
   431→                </div>
   432→              </div>
   433→            ))}
   434→          </div>
   435→        )}
   436→      </CardContent>
   437→    </Card>
